cmake_minimum_required(VERSION 3.16)
project(geany_insight C)
set(CMAKE_C_STANDARD 99)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS vendor/lua/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac|onelua|linit|liolib|loslib|loadlib|lcorolib|ldblib)\\.c$")
add_library(insight_core STATIC core/runtime.c core/index.c core/document.c ${LUA_SOURCES})
set_target_properties(insight_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(insight_core PUBLIC core PRIVATE vendor/lua)
if(UNIX)
 target_link_libraries(insight_core PUBLIC m)
endif()
option(INSIGHT_GEANY_PLUGIN "Build the desktop Geany adapter" OFF)
if(INSIGHT_GEANY_PLUGIN)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(GEANY REQUIRED IMPORTED_TARGET geany)
 add_library(insight_lua MODULE desktop/plugin.c desktop/navigation.c)
 target_link_libraries(insight_lua PRIVATE insight_core PkgConfig::GEANY)
 set_target_properties(insight_lua PROPERTIES PREFIX "")
endif()
if(ANDROID)
 add_library(insight_android SHARED android/jni.c)
 target_link_libraries(insight_android PRIVATE insight_core)
endif()
if(NOT ANDROID)
 enable_testing()
 add_executable(runtime_test tests/runtime_test.c)
 target_link_libraries(runtime_test PRIVATE insight_core)
 add_test(NAME runtime COMMAND runtime_test)
 add_executable(index_test tests/index_test.c)
 target_link_libraries(index_test PRIVATE insight_core)
 add_test(NAME index COMMAND index_test)
 add_executable(document_test tests/document_test.c)
 target_link_libraries(document_test PRIVATE insight_core)
 add_test(NAME document COMMAND document_test)
 if(MSVC)
  target_compile_options(runtime_test PRIVATE /UNDEBUG)
  target_compile_options(index_test PRIVATE /UNDEBUG)
  target_compile_options(document_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(runtime_test PRIVATE -UNDEBUG)
  target_compile_options(index_test PRIVATE -UNDEBUG)
  target_compile_options(document_test PRIVATE -UNDEBUG)
 endif()
endif()
